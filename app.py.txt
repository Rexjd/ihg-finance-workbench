# ========== 4.GOP情景测算 ==========
elif menu == "📉 GOP情景测算":
    st.header("📉 GOP多情景测算（经营决策）")
    col1, col2 = st.columns(2)
    with col1:
        occ = st.slider("客房出租率 %", 40, 95, 78)
        adr = st.slider("平均房价 ADR", 350, 700, 525)
        market_pct = st.slider("市场费用占收入比 %", 3, 12, 7)
        labor_pct = st.slider("人力成本占收入比 %", 22, 35, 28)
    with col2:
        room_count = 240
        room_revenue = occ / 100 * room_count * adr * 30
        market_cost = room_revenue * market_pct /100
        labor_cost = room_revenue * labor_pct /100
        other_cost = room_revenue * 0.22
        gop = room_revenue - market_cost - labor_cost - other_cost
        gop_pct = gop / room_revenue

        st.metric("月度客房收入", f"¥{room_revenue:,.0f}")
        st.metric("市场费用", f"¥{market_cost:,.0f}")
        st.metric("人力成本", f"¥{labor_cost:,.0f}")
        st.metric("测算GOP", f"¥{gop:,.0f}")
        st.metric("测算GOP%", f"{gop_pct:.1%}")
        st.info("用于总经理会：模拟调价、人力优化、营销投入变化对利润的影响")
